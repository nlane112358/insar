cmake_minimum_required(VERSION 3.18)
project(insar_cpu_proto LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(CheckCXXCompilerFlag)

if(APPLE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    foreach(_omp_prefix IN ITEMS /opt/homebrew/opt/libomp /usr/local/opt/libomp)
        if(EXISTS "${_omp_prefix}/include/omp.h")
            list(APPEND CMAKE_PREFIX_PATH "${_omp_prefix}")
            break()
        endif()
    endforeach()
endif()

find_package(OpenMP)

add_executable(insar_proto src/insar_engine_cpu.cpp)

if(OpenMP_CXX_FOUND)
    target_link_libraries(insar_proto PRIVATE OpenMP::OpenMP_CXX)
endif()

if(MSVC)
    target_compile_options(insar_proto PRIVATE /O2 /arch:AVX2)
else()
    target_compile_options(insar_proto PRIVATE -O3 -ffast-math)
    if(APPLE OR CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64|ARM64)")
        check_cxx_compiler_flag("-mcpu=native" HAS_MCPU_NATIVE)
        if(HAS_MCPU_NATIVE)
            target_compile_options(insar_proto PRIVATE -mcpu=native)
        else()
            check_cxx_compiler_flag("-march=native" HAS_MARCH_NATIVE)
            if(HAS_MARCH_NATIVE)
                target_compile_options(insar_proto PRIVATE -march=native)
            endif()
        endif()
    else()
        check_cxx_compiler_flag("-march=native" HAS_MARCH_NATIVE)
        if(HAS_MARCH_NATIVE)
            target_compile_options(insar_proto PRIVATE -march=native)
        else()
            check_cxx_compiler_flag("-mcpu=native" HAS_MCPU_NATIVE)
            if(HAS_MCPU_NATIVE)
                target_compile_options(insar_proto PRIVATE -mcpu=native)
            endif()
        endif()
    endif()
endif()
